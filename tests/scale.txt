# SCALE + one-finger chords (ui/ui_scale.c). Numeric key ids:
# 0=F3 OCT-, 7..18 = C4..B4 roots, 19..26 = C5..G5 = MAJ MIN DIM 6 7 M7 SUS4 ADD9.
# Readbacks (app_expect): scale = kind<<8|tonic, ofc, ofc_mask = mask<<8|root,
# step0 = nvoices<<8|first-note. (The DSL's expect compare is broken — "== ==" —
# so the FAIL lines still print the true value; the outs are the real assertions.)
wait 600
turn SELECT 3                     # the demo's bass track (ch 0, melodic)
tap SEL                           # the SCALE page opens
turn K3 1                         # OFC on
tap SEQ                           # back to the GRID
# root alone: sounds, jams, does NOT write the step
key 7 down 100                    # C4
wait 20
key 7 up
wait 20
# root + MAJ = C major triad, written at cursor 0, cursor advances once
key 7 down 100
wait 20
key 19 down 100                   # MAJ
wait 20
out
expect ofc_mask == 0x107
expect step0 == 0x33C
key 19 up
key 7 up
wait 20
out
# sus4 swap while the root is held: the old third turns off (the diff apply)
key 7 down 100
wait 20
key 19 down 100
wait 20
key 19 up
key 25 down 100                   # SUS4: 60 65 67, the E (64) off
wait 20
out
key 25 up
key 7 up
wait 20
out
# F major 9 = root 13 + MAJ + 7 + ADD9: five tones drop the fifth
key 13 down 100
wait 20
key 19 down 100
wait 20
key 22 down 100                   # 7
wait 20
key 26 down 100                   # ADD9
wait 30
out
key 26 up
key 22 up
key 19 up
key 13 up
wait 20
out
# nonsense set: 7 + M7 together = no tones (silent), state still tracked
key 10 down 100                   # D#4 -> snaps to D4 (62) in C major
wait 20
key 22 down 100
wait 20
key 23 down 100                   # 7 + M7 = nonsense
wait 30
expect ofc_mask == 0x180A
key 23 up
key 22 up
key 10 up
wait 20
out

# the DEGREE walk (left 7 keys, I..VII): chord follows the scale, key up returns
# (root C4 + MAJ held throughout: C maj -> IV -> ii -> back)
key 7 down 100
key 19 down 100
wait 20
out                              # C major: 60 64 67
key 3 down 100                   # IV: Fmaj7 (the scale's own seventh) 65 69 72 76
wait 20
out
key 3 up
key 1 down 100                   # ii with the MAJ flavor: D F# A (62 66 69)
wait 20
out
key 1 up
wait 20
out                              # back to C major
key 19 up
key 7 up
wait 20
