# GLOBAL page (GLO+HOME): tempo, both swing halves, the GM switch, Bluetooth.
#
# The page's contract in one line: everything here is one number for the whole
# instrument, it is rare enough to live behind a chord, and nothing on it
# records into the jam ring.
#
# Geometry: 5 rows at y30 + 24*i, labels x12 (FONT_S), values x100 (FONT_B).
# Door: GLO+HOME on the press edge (GLO is latched by ui_input, never dispatched).
# Leave: a bare HOME, the same gesture every page uses.

wait 600
expect mode 0
expect bpm 120
expect gl_gm 1
expect gm_alloff 0

# ---- the door ----
press GLO
press HOME
release HOME
release GLO
wait 20
expect mode 5
expect gl_row 0
# HOME while the page is up must not also toggle HOME/GRID underneath it
tap HOME
wait 10
expect mode 0
# and the chord works from anywhere, not only from HOME
tap SEQ
wait 10
expect mode 1
press GLO
press HOME
release HOME
release GLO
wait 20
expect mode 5

# ---- tempo: 1 BPM a detent, 10 with GLO, dead stop at both ends ----
turn K1 0
turn K2 5
wait 5
expect bpm 125
press GLO
turn K2 3
release GLO
wait 5
expect bpm 155
turn K2 -1
wait 5
expect bpm 154
turn K2 200
wait 5
expect bpm 300
turn K2 -400
wait 5
expect bpm 20
turn K2 100
wait 5
expect bpm 120

# ---- tap tempo on the rightmost white key (key 26, G5) ----
# tapkey itself runs 90 ms (60 down, 30 up), so a 500 ms interval needs a 410 wait.
tapkey 26 100
wait 410
tapkey 26 100
wait 410
tapkey 26 100
wait 410
tapkey 26 100
wait 410
tapkey 26 100
wait 50
expect bpm 120
# 375 ms apart is 160 BPM, and the average follows the new tempo (the pause
# before it is longer than GL_TAP_GAP, so the old taps are dropped)
wait 2500
tapkey 26 100
wait 285
tapkey 26 100
wait 285
tapkey 26 100
wait 285
tapkey 26 100
wait 50
expect bpm 160
# a hand-set tempo ends the tap measurement: one detent is one BPM, not a jump
# back to some earlier average
turn K2 1
wait 5
expect bpm 161
turn K2 -41
wait 5
expect bpm 120

# ---- swing: the two halves, and the row that says they are one number ----
turn K1 1
wait 5
expect gl_row 1
turn K2 20
wait 5
expect swing_global 20
press GLO
wait 5
turn K2 3
wait 5
release GLO
wait 5
expect swing_global 50
turn K2 50
wait 5
expect swing_global 100        # clamped
turn K2 -150
wait 5
expect swing_global 0

# ---- the per-track half, with the track it belongs to ----
# The swing block above left the row on SWING; one more detent is TRK SWING.
# The track is still T1 (K3 is the only thing on this page that moves it).
turn K1 1
wait 5
expect gl_row 2
expect sel_track 0
expect swing_track 100          # what the demo left
# `turn Kn N` is N detents, each 10 with GLO and 1 without: so -10 with GLO is
# -100, and the second turn is what makes the assertion about the step size and
# not about the fixture.
press GLO
wait 5
turn K2 -1
wait 5
expect swing_track 90          # one detent is TEN here, not one
turn K2 -9
wait 5
expect swing_track 0
release GLO
wait 5
turn K2 30
wait 5
expect swing_track 30
# K3 picks the track, and the row follows it: two tracks have two weights, and
# editing one must not touch the other. T2's value is whatever the demo left
# (100), so it is read first and then moved, and the check is that T1 kept 30.
turn K3 1
wait 5
expect sel_track 1
expect swing_track 100
turn K2 5
wait 5
expect swing_track 100          # already clamped at the top
press GLO
wait 5
turn K2 -5
wait 5
release GLO
wait 5
expect swing_track 50
turn K3 -1
wait 5
expect sel_track 0
expect swing_track 30           # T1 untouched by T2's edit
# dead stop at the last row
turn K1 5
wait 5
expect gl_row 4

# ---- the GM switch: CC123 on the way out, then the feed closes ----
turn K1 -2
wait 5
expect gl_row 2
turn K1 1
wait 5
expect gl_row 3
turn K2 1
wait 5
expect gl_gm 0
expect gm_alloff 1
# zero the window, run a full pattern: nothing reaches the synth
expect gm_reset 0
tap PLAY
wait 700
tap PLAY
wait 100
expect gm_noteons 0
# back on: the same run does reach it
turn K2 1
wait 5
expect gl_gm 1
expect gm_reset 0
tap PLAY
wait 700
tap PLAY
wait 100
expect gm_noteons >0

# ---- keys on this page audition and never record ----
turn K1 -3
wait 5
expect gl_row 0
expect jam_used 0
key 5 down
wait 60
key 5 up
wait 40
key 26 down
wait 40
key 26 up
wait 40
expect jam_used 0

# ---- leaving writes the settings ----
expect store_writes 0
tap HOME
wait 20
expect mode 0
expect store_writes >0
